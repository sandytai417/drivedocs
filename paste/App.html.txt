  /** 前端自算週次（免打伺服器） */
  function listWeekOptionsClient_(weeksBack) {
    weeksBack = weeksBack == null ? 16 : weeksBack;
    var now = new Date();
    var ymd = taipeiYmdParts_(now);
    var local = new Date(ymd.y, ymd.m - 1, ymd.d);
    var isoDow = taipeiIsoDow_(now);
    var thisMonday = new Date(local);
    thisMonday.setDate(local.getDate() - (isoDow - 1));
    var pad = function (n) { return (n < 10 ? '0' : '') + n; };
    var fmt = function (x) {
      return x.getFullYear() + '/' + pad(x.getMonth() + 1) + '/' + pad(x.getDate());
    };
    var out = [];
    for (var w = 0; w <= weeksBack; w++) {
      var monday = new Date(thisMonday);
      monday.setDate(thisMonday.getDate() - w * 7);
      var sunday = new Date(monday);
      sunday.setDate(monday.getDate() + 6);
      var value = fmt(monday) + ' – ' + fmt(sunday);
      var tag = w === 0 ? '（本週）' : (w === 1 ? '（上週）' : ('（' + w + ' 週前）'));
      out.push({ value: value, label: value + ' ' + tag, offset: w, isCurrent: w === 0 });
    }
    return out;
  }

  function taipeiYmdParts_(date) {
    try {
      var parts = new Intl.DateTimeFormat('en-CA', {
        timeZone: 'Asia/Taipei', year: 'numeric', month: '2-digit', day: '2-digit'
      }).formatToParts(date);
      var map = {};
      parts.forEach(function (p) { map[p.type] = p.value; });
      return { y: Number(map.year), m: Number(map.month), d: Number(map.day) };
    } catch (e) {
      return { y: date.getFullYear(), m: date.getMonth() + 1, d: date.getDate() };
    }
  }

  function taipeiIsoDow_(date) {
    try {
      var wd = new Intl.DateTimeFormat('en-US', { timeZone: 'Asia/Taipei', weekday: 'short' }).format(date);
      var map = { Mon: 1, Tue: 2, Wed: 3, Thu: 4, Fri: 5, Sat: 6, Sun: 7 };
      return map[wd] || 1;
    } catch (e) {
      var js = date.getDay();
      return js === 0 ? 7 : js;
    }
  }

  function mondayFromPeriodClient_(period) {
    var s = String(period || '').trim();
    var m = s.match(/(\d{4})[\/\-](\d{1,2})[\/\-](\d{1,2})/);
    if (m) return new Date(Number(m[1]), Number(m[2]) - 1, Number(m[3]));
    var ymd = taipeiYmdParts_(new Date());
    var local = new Date(ymd.y, ymd.m - 1, ymd.d);
    var iso = taipeiIsoDow_(new Date());
    local.setDate(local.getDate() - (iso - 1));
    return local;
  }

  /** 瀏覽器原生編碼，比自行 btoa 迴圈快 */
  function readFileAsBase64_(file) {
    return new Promise(function (resolve, reject) {
      var reader = new FileReader();
      reader.onload = function () {
        var s = String(reader.result || '');
        var i = s.indexOf(',');
        resolve(i >= 0 ? s.slice(i + 1) : s);
      };
      reader.onerror = function () { reject(reader.error || new Error('讀檔失敗')); };
      reader.readAsDataURL(file);
    });
  }

  function isCompressibleImage_(file) {
    var t = String((file && file.type) || '').toLowerCase();

  /** 大張照片先縮小，減少讀檔與上傳時間（保單掃描仍用原檔 PDF） */
  function compressImageFile_(file) {
    return new Promise(function (resolve) {
      if (!file || !isCompressibleImage_(file) || file.size < 450000) {
        resolve(file);
        return;
      }
      var url = URL.createObjectURL(file);
      var img = new Image();
      img.onload = function () {
        try {
          var maxEdge = 1920;
          var w = img.naturalWidth || img.width || 0;
          var h = img.naturalHeight || img.height || 0;
          var scale = Math.min(1, maxEdge / Math.max(w, h, 1));
          var cw = Math.max(1, Math.round(w * scale));
          var ch = Math.max(1, Math.round(h * scale));
          var canvas = document.createElement('canvas');
          canvas.width = cw;
          canvas.height = ch;
          var ctx = canvas.getContext('2d');
          if (!ctx) {
            URL.revokeObjectURL(url);
            resolve(file);
            return;
          }
          ctx.drawImage(img, 0, 0, cw, ch);
          URL.revokeObjectURL(url);
          canvas.toBlob(function (blob) {
            if (!blob || blob.size >= file.size * 0.92) {
              resolve(file);
              return;
            }
            var name = String(file.name || 'photo.jpg').replace(/\.(png|webp|jpeg|jpg)$/i, '.jpg');
            if (!/\.jpg$/i.test(name)) name += '.jpg';
            try {
              resolve(new File([blob], name, { type: 'image/jpeg' }));
            } catch (e2) {
              resolve(blob);
            }
          }, 'image/jpeg', 0.82);
        } catch (e) {
          try { URL.revokeObjectURL(url); } catch (e2) { /* ignore */ }
          resolve(file);
        }
      };
      img.onerror = function () {
        try { URL.revokeObjectURL(url); } catch (e) { /* ignore */ }
        resolve(file);
      };
      img.src = url;
    });
  }

  async function mapPool_(items, limit, fn) {
    var out = new Array(items.length);
    var cursor = 0;
    async function worker_() {
      while (cursor < items.length) {
        var idx = cursor;
        cursor += 1;
        out[idx] = await fn(items[idx], idx);
      }
    }
    var n = Math.max(1, Math.min(limit || 2, items.length));
    var workers = [];
    for (var w = 0; w < n; w++) workers.push(worker_());
    await Promise.all(workers);
    return out;
  }

  /** 西元 yyyy-MM-dd → 069/08/08；MM-dd → 08/08 */
  function formatRocDateClient_(ymd) {
    var s = String(ymd == null ? '' : ymd).trim();
    if (!s) return '';
    var m = s.match(/^(\d{4})-(\d{2})-(\d{2})$/);
    if (m) {
      var ry = rocYearFromGregorianClient_(m[1]);
      if (!(ry >= 1 && ry <= 999)) return s;
      var ryStr = ry >= 100 ? String(ry) : ('00' + ry).slice(-3);
      return ryStr + '/' + m[2] + '/' + m[3];
    }
    m = s.match(/^(\d{1,2})-(\d{1,2})$/);
    if (m) return pad2Client_(m[1]) + '/' + pad2Client_(m[2]);
    return s;
  }

  /**
   * 日期輸入 → 西元 yyyy-MM-dd（資料夾／儲存用）
   * 民國：690808、0690808、1150806、69/08/08、民國69年8月8日
   * 西元仍相容：19800808、1980-08-08
   */
  function parseDateInputToYmdClient_(raw, emptyDefault) {
    var s = String(raw == null ? '' : raw).trim();
    if (!s) {
      return emptyDefault === undefined ? todayLocalYmd_() : emptyDefault;
    }
    s = s.replace(/^民國\s*/i, '').replace(/号/g, '日');

    var m = s.match(/^(\d{1,3})[\/\-.\s年](\d{1,2})[\/\-.\s月](\d{1,2})日?$/);
    if (m && Number(m[1]) >= 1 && Number(m[1]) <= 999) {
      var gy = gregorianYearFromRocClient_(m[1]);
      if (gy >= 1912 && gy <= 2100) {
        return gy + '-' + pad2Client_(m[2]) + '-' + pad2Client_(m[3]);
      }
    }
    m = s.match(/^(\d{4})[\/\-.\s年](\d{1,2})[\/\-.\s月](\d{1,2})日?$/);
    if (m) {
      return m[1] + '-' + pad2Client_(m[2]) + '-' + pad2Client_(m[3]);
    }

    var digits = s.replace(/\D/g, '');
    if (digits.length === 8) {
      return digits.slice(0, 4) + '-' + digits.slice(4, 6) + '-' + digits.slice(6, 8);
    }
    if (digits.length === 7) {
      return gregorianYearFromRocClient_(digits.slice(0, 3)) + '-' + digits.slice(3, 5) + '-' + digits.slice(5, 7);
    }
    if (digits.length === 6) {
      return gregorianYearFromRocClient_(digits.slice(0, 2)) + '-' + digits.slice(2, 4) + '-' + digits.slice(4, 6);
    }
    if (digits.length === 4) {
      var y = String(emptyDefault || todayLocalYmd_()).slice(0, 4);
      return y + '-' + digits.slice(0, 2) + '-' + digits.slice(2, 4);
    }
    return s;
  }

  /** 試算表 Date／英文時間字串 → yyyy-MM-dd */
  function coerceDateLikeToYmdClient_(raw) {
    if (raw == null || raw === '') return '';
    if (Object.prototype.toString.call(raw) === '[object Date]' && !isNaN(raw.getTime())) {
      try {
        return new Intl.DateTimeFormat('en-CA', {
          timeZone: 'Asia/Taipei', year: 'numeric', month: '2-digit', day: '2-digit'
        }).format(raw);
      } catch (e0) {
        return raw.getFullYear() + '-' + pad2Client_(raw.getMonth() + 1) + '-' + pad2Client_(raw.getDate());
      }
    }
    var s = String(raw).trim();
    if (!s) return '';
    if (/^\d{4}-\d{2}-\d{2}/.test(s)) return s.slice(0, 10);
    // Sat Apr 24 1982 00:00:00 GMT+0800 (台北標準時間) — 直接抓月日，避免時區差一天
    var eng = s.match(/\b([A-Za-z]{3})\s+(\d{1,2})\s+(\d{4})\b/);
    if (eng) {
      var months = {
        Jan: 1, Feb: 2, Mar: 3, Apr: 4, May: 5, Jun: 6,
        Jul: 7, Aug: 8, Sep: 9, Oct: 10, Nov: 11, Dec: 12
      };
      var mm = months[eng[1]];
      if (mm) return eng[3] + '-' + pad2Client_(mm) + '-' + pad2Client_(eng[2]);
    }
    if (/GMT|UTC|標準時間/i.test(s)) {
      var d = new Date(s);
      if (!isNaN(d.getTime())) {
        try {
          return new Intl.DateTimeFormat('en-CA', {
            timeZone: 'Asia/Taipei', year: 'numeric', month: '2-digit', day: '2-digit'
          }).format(d);
        } catch (e1) {
          return d.getFullYear() + '-' + pad2Client_(d.getMonth() + 1) + '-' + pad2Client_(d.getDate());
        }
      }
    }
    return s;
  }

  /**
   * 生日 → 儲存用西元 yyyy-MM-dd（或僅月日 MM-dd）
   */
  function normalizeBirthdayClient_(raw) {
    var coerced = coerceDateLikeToYmdClient_(raw);
    if (/^\d{4}-\d{2}-\d{2}$/.test(coerced)) return coerced;
    var s = String(coerced || raw || '').trim();
    if (!s) return '';
    s = s.replace(/^民國\s*/i, '');
    // 先解析完整日期，避免「69年8月8日」被誤判成 4 碼月日
    var ymd = parseDateInputToYmdClient_(s, '');
    if (/^\d{4}-\d{2}-\d{2}$/.test(ymd)) return ymd;
    var digits = s.replace(/\D/g, '');
    if (digits.length === 8) {
      return digits.slice(0, 4) + '-' + digits.slice(4, 6) + '-' + digits.slice(6, 8);
    }
    if (digits.length === 7) {
      return gregorianYearFromRocClient_(digits.slice(0, 3)) + '-' + digits.slice(3, 5) + '-' + digits.slice(5, 7);
    }
    if (digits.length === 6) {
      return gregorianYearFromRocClient_(digits.slice(0, 2)) + '-' + digits.slice(2, 4) + '-' + digits.slice(4, 6);
    }
    var m = s.match(/^(\d{1,2})[\/\-.\s月](\d{1,2})日?$/);
    if (m) return pad2Client_(m[1]) + '-' + pad2Client_(m[2]);
    if (digits.length === 4) {
      return digits.slice(0, 2) + '-' + digits.slice(2, 4);
    }
    return s;
  }

  /** 生日顯示：民國69年8月8日 */
  function formatBirthdayDisplayClient_(raw) {
    var stored = normalizeBirthdayClient_(raw);
    if (!stored) return '';
    var m = String(stored).match(/^(\d{4})-(\d{2})-(\d{2})$/);
    if (m) {
      var ry = rocYearFromGregorianClient_(m[1]);
      if (ry >= 1 && ry <= 999) {
        return '民國' + ry + '年' + Number(m[2]) + '月' + Number(m[3]) + '日';
      }
    }
    m = String(stored).match(/^(\d{1,2})-(\d{1,2})$/);
    if (m) return Number(m[1]) + '月' + Number(m[2]) + '日';
    return stored;
  }

    var page = state.page;

      if (!content.innerHTML || !String(content.innerHTML).trim()) {
        content.innerHTML = '<p class="muted soft-load" style="padding:24px">載入中…</p>';
      }
    }
    try {
      if (page === 'dashboard') await renderDashboard(content);
      else if (page === 'customers') await renderCustomers(content);
      else if (page === 'upload') await renderUpload(content);
      else if (page === 'activities') await renderEvents(content);
      else if (page === 'settings') await renderSettings(content);
      else if (page === 'detail') await renderDetail(content);
      else if (page === 'search') await renderSearch(content);
      if (seq !== renderSeq_) return;
      content.classList.remove('is-loading');
      animateBars(content);
    } catch (err) {
      if (seq !== renderSeq_) return;
      content.classList.remove('is-loading');
      content.innerHTML = '<div class="card card__bd"><p>' + escapeHtml(err.message || err) + '</p></div>';
      toast(err.message || String(err), true);
    }
  }

  async function runUploadFlow_(btn) {
    var queue = state.uploadQueue;
    var total = queue.length;
    await setUploadProgress_({ label: '準備中', percent: 2, done: 0, total: total });

    if (state.uploadTarget === 'activity') {
      var idxAll = {};
      queue.forEach(function (_, k) { idxAll[k] = true; });
      var localOnly = {};
      queue.forEach(function (it, k) {
        if (it.source !== 'drive') localOnly[k] = true;
      });
      if (Object.keys(localOnly).length) markQueue_(localOnly, '讀檔／壓縮', 15);

      var prepared = 0;
      var files = await mapPool_(queue, 2, async function (it) {
        var out;
        if (it.source === 'drive') {
          out = { name: it.name, fileId: it.driveFileId, driveFileId: it.driveFileId,
                   size: it.size, mimeType: it.mimeType };
        } else {
          var p = await prepareLocalFile_(it.file);
          out = { name: p.fileName, size: p.size, mimeType: p.mimeType, base64Data: p.base64Data };
        }
        prepared += 1;
        var pct = 8 + Math.round((prepared / Math.max(1, total)) * 52);
        await setUploadProgress_({
          label: (it.source === 'drive' ? '雲端檔案 ' : '讀檔／壓縮 ') + prepared + '/' + total,
          percent: pct,
          done: prepared,
          total: total
        });
        return out;
      });
      await setUploadProgress_({ label: '上傳到本週活動', percent: 70, done: 0, total: total });
      markQueue_(idxAll, '上傳中', 70);

      var created = await run('api_quickAddToThisWeekActivity', {
        files: files,
        importMode: (queue[0] && queue[0].importMode) || 'move'
      });
      markQueue_(idxAll, '完成', 100);
      await setUploadProgress_({
        label: '上傳成功',
        percent: 100,
        done: total,
        total: total,
        success: true
      });
      toast('上傳成功');
      await sleepMs_(1200);
      state.uploadQueue = [];
      state.cache.activities = null; state.cache.activitiesAt = 0;
      await setPage('activities');
      await setUploadProgress_(null);
      return;
    }

    // === 客戶資料夾 ===
    var form = readCustomerForm_();
    if (!form.customerId && !form.name) {
      throw new Error('請填寫姓名，或從上方選擇既有客戶');
    }
    if (btn) btn.textContent = '建立客戶…';
    await setUploadProgress_({ label: '準備客戶', percent: 5, done: 0, total: total });
    var ensured = await run('api_ensureCustomer', {
      customerId: form.customerId || undefined,
      id: form.customerId || undefined,
      customerName: form.name, name: form.name,
      phone: form.phone, email: form.email, birthday: form.birthday,
      gender: form.gender, idNumber: form.idNumber, address: form.address
    });
    invalidateClientCache_();
    var customerId = ensured.id;
    state.uploadCustomerId = customerId;

    var docDateRaw = ($('n-doc-date') && $('n-doc-date').value) || state.uploadDocDate || todayLocalYmdDigits_();
    state.uploadDocDate = String(docDateRaw).trim();
    var docDate = normalizeDocDateClient_(state.uploadDocDate);

    var drivePos = [], localPos = [];
    for (var i = 0; i < queue.length; i++) {
      (queue[i].source === 'drive' ? drivePos : localPos).push(i);
    }

    var done = 0;
    async function tick_(delta, label) {
      done += (delta || 0);
      var pct = total ? Math.round((done / total) * 90) + 8 : 100;
      await setUploadProgress_({ label: label || '上傳中', percent: Math.min(98, pct), done: done, total: total });
    }

    // (1) 雲端匯入：一次呼叫全部
    if (drivePos.length) {
      var idxSet = {}; drivePos.forEach(function (k) { idxSet[k] = true; });
      markQueue_(idxSet, '匯入中', 40);
      if (btn) btn.textContent = '匯入雲端 ' + drivePos.length + ' 個…';
      await setUploadProgress_({ label: '匯入雲端檔案', percent: 20, done: done, total: total });
      var res = await run('api_importDocuments', {
        customerId: customerId,
        items: drivePos.map(function (k) {
          var it = queue[k];
          return {
            category: it.docType || state.uploadCategory,
            fileId: it.driveFileId,
            fileName: it.name,
            mode: it.importMode || 'move',
            docDate: docDate
          };
        })
      });
      applyBatchResult_(res, drivePos);
      await tick_(drivePos.length, '雲端匯入完成');
    }

    // (2) 本機檔案：一次只讀一批，讀下一批與上傳重疊
    if (localPos.length) {
      var localItems = localPos.map(function (k) {
        var it = queue[k];
        return {
          _pos: k,
          file: it.file,
          size: Number(it.size) || 0,
          category: it.docType || state.uploadCategory
        };
      });
      var batches = chunkBySize_(localItems, 8 * 1024 * 1024);
      var readDone = 0;

      async function prepareBatch_(batch) {
        var idxSet2 = {};
        batch.forEach(function (x) { idxSet2[x._pos] = true; });
        markQueue_(idxSet2, '讀檔／壓縮', 20);
        if (btn) btn.textContent = '讀檔／壓縮…';
        return mapPool_(batch, 2, async function (x) {
          var p = await prepareLocalFile_(x.file);
          readDone += 1;
          var readPct = 12 + Math.round((readDone / Math.max(1, localPos.length)) * 28);
          await setUploadProgress_({
            label: '讀檔／壓縮 ' + readDone + '/' + localPos.length,
            percent: Math.min(48, readPct),
            done: done,
            total: total
          });
          return {
            _pos: x._pos,
            category: x.category,
            fileName: p.fileName,
            mimeType: p.mimeType,
            base64Data: p.base64Data,
            docDate: docDate
          };
        });
      }

      var pending = batches.length ? prepareBatch_(batches[0]) : null;
      for (var b = 0; b < batches.length; b++) {
        var payloadItems = await pending;
        pending = (b + 1 < batches.length) ? prepareBatch_(batches[b + 1]) : null;
        var posInBatch = payloadItems.map(function (x) { return x._pos; });
        var batchIdx = {};
        posInBatch.forEach(function (k) { batchIdx[k] = true; });
        markQueue_(batchIdx, '上傳中', 65);
        if (btn) btn.textContent = '上傳中 ' + (b + 1) + '/' + batches.length;
        await setUploadProgress_({
          label: '上傳中 ' + (b + 1) + '/' + batches.length,
          percent: 55 + Math.round(((b) / Math.max(1, batches.length)) * 40),
          done: done,
          total: total
        });
        var sendItems = payloadItems.map(function (x) {
          return {
            category: x.category,
            fileName: x.fileName,
            mimeType: x.mimeType,
            base64Data: x.base64Data,
            docDate: x.docDate
          };
        });
        var r = await run('api_uploadDocuments', { customerId: customerId, items: sendItems });
        applyBatchResult_(r, posInBatch);
        await tick_(posInBatch.length, '已上傳 ' + done + '/' + total);
      }
    }

    await setUploadProgress_({ label: '上傳成功', percent: 100, done: total, total: total, success: true });
    toast('上傳成功');
    await sleepMs_(1200);
    state.uploadQueue = [];
    state.detailCache = null;
    invalidateClientCache_();
    if (btn) btn.textContent = '開啟客戶…';
    await setPage('detail', { customerId: customerId, tab: 'folders' });
    await setUploadProgress_(null);
  }

  function markQueue_(idxSet, status, progress) {
    for (var k = 0; k < state.uploadQueue.length; k++) {
      if (idxSet[k]) {
        state.uploadQueue[k].status = status;
        if (progress != null) state.uploadQueue[k].progress = progress;
      }
    }
    $('queue-box').innerHTML = renderQueue();
    bindQueue();
  }

  function chunkBySize_(items, maxBytes) {
    var out = [];
    var buf = [];
    var size = 0;
    for (var i = 0; i < items.length; i++) {
      var s = Number(items[i].size) || 0;
      if (buf.length && (size + s) > maxBytes) {
        out.push(buf); buf = []; size = 0;
      }
      buf.push(items[i]);
      size += s;
    }
    if (buf.length) out.push(buf);
    return out;
  }

  function applyBatchResult_(res, positions) {
    var results = (res && res.results) || [];
    for (var i = 0; i < positions.length; i++) {
      var pos = positions[i];
      var r = results[i];
      if (!state.uploadQueue[pos]) continue;
      if (r && r.ok) {
        state.uploadQueue[pos].status = '完成';
        state.uploadQueue[pos].progress = 100;
      } else {
        state.uploadQueue[pos].status = '失敗: ' + ((r && r.error) || '未知');
        state.uploadQueue[pos].progress = 0;
      }
    }
    $('queue-box').innerHTML = renderQueue();
    bindQueue();
  }

  function tryOpenGooglePicker_(onDone) {
    return new Promise(function (resolve, reject) {
      function fail(err) { reject(err || new Error('picker unavailable')); }

      function startPicker(cfg) {
        if (typeof google === 'undefined' || !google.picker) {
          fail();
          return;
        }
        try {
          var view = new google.picker.DocsView(google.picker.ViewId.DOCS)
            .setIncludeFolders(true)
            .setSelectFolderEnabled(false)
            .setMode(google.picker.DocsViewMode.LIST);
          var builder = new google.picker.PickerBuilder()
            .addView(view)
            .addView(google.picker.ViewId.RECENTLY_PICKED)
            .enableFeature(google.picker.Feature.MULTISELECT_ENABLED)
            .enableFeature(google.picker.Feature.SUPPORT_DRIVES)
            .setOAuthToken(cfg.token)
            .setOrigin(cfg.origin || 'https://script.google.com')
            .setTitle('選取雲端硬碟檔案')
            .setCallback(function (data) {
              if (!data) return;
              if (data.action === google.picker.Action.CANCEL) return;
              if (data.action !== google.picker.Action.PICKED) return;
              var docs = data.docs || [];
              var files = docs.filter(function (d) {
                return d && d.id && d.mimeType !== 'application/vnd.google-apps.folder';
              }).map(function (d) {
                return {
                  id: d.id,
                  name: d.name,
                  mimeType: d.mimeType || '',
                  size: d.sizeBytes || 0,
                  url: d.url || ''
                };
              });
              if (!files.length) {
                toast('請選擇檔案（不要選資料夾）', true);
                return;
              }
              closeModal();
              onDone(files, 'move');
              toast('已加入 ' + files.length + ' 個雲端檔案（將移動）');
            });
          var picker = builder.build();
          picker.setVisible(true);
          resolve(true);
        } catch (e) {
          fail(e);
        }
      }

      loadGapiScript_().then(function () {
        return run('api_getPickerConfig');
      }).then(function (cfg) {
        if (!cfg || !cfg.token) {
          fail();
          return;
        }
        if (typeof gapi === 'undefined' || !gapi.load) {
          fail(new Error('尚未載入 Google API'));
          return;
        }
        gapi.load('picker', {
          callback: function () { startPicker(cfg); },
          onerror: fail,
          timeout: 8000,
          ontimeout: fail
        });
      }).catch(fail);
    });
  }

  function loadGapiScript_() {
    return new Promise(function (resolve, reject) {
      if (typeof gapi !== 'undefined' && gapi.load) {
        resolve();
        return;
      }
      var existing = document.getElementById('gapi-js');
      if (existing) {
        existing.addEventListener('load', function () { resolve(); });
        existing.addEventListener('error', reject);
        return;
      }
      var s = document.createElement('script');
      s.id = 'gapi-js';
      s.src = 'https://apis.google.com/js/api.js';
      s.async = true;
      s.onload = function () { resolve(); };
      s.onerror = reject;
      document.head.appendChild(s);
    });
  }

      html += '<ul class="queue" style="margin:0">';

        html += '<li class="queue__item" style="padding:10px 0;border-bottom:1px solid rgba(13,13,13,.08)">' +
          '<label style="display:flex;gap:10px;align-items:flex-start;cursor:pointer">' +

          loadFolder(meta.parentId || '');
        };

        btn.onclick = function () { loadFolder(btn.getAttribute('data-id')); };

  function applyUpdatedCustomer_(updated) {
    if (!updated || !updated.id) return;
    updated._fullInfo = true;
    if (state.detailCache && state.detailCache.customer &&
        String(state.detailCache.customer.id) === String(updated.id)) {
      state.detailCache.customer = Object.assign({}, state.detailCache.customer, updated, { _fullInfo: true });
    }
    function patchList_(list) {
      if (!list || !list.customers) return;
      for (var i = 0; i < list.customers.length; i++) {
        if (String(list.customers[i].id) === String(updated.id)) {
          list.customers[i] = Object.assign({}, list.customers[i], updated);
        }
      }
    }
    patchList_(state.cache.customers);
    if (state.cache.home && state.cache.home.customers) patchList_(state.cache.home.customers);
  }

  async function hydrateDetailCustomer_(data, seq) {
    if (!data || !data.customer || data.customer._fullInfo) return data;
    var full = await run('api_getCustomer', data.customer.id, false);
    if (seq != null && seq !== renderSeq_) return data;
    if (full && full.customer) {
      data.customer = Object.assign({}, data.customer, full.customer, { _fullInfo: true });
      if (full.folderMeta) data.folderMeta = full.folderMeta;
      state.detailCache = data;
    }
    return data;
  }

  function findCachedCustomer_(id) {
    var list = state.cache.customers;
    if (!list || !list.customers) return null;
    for (var i = 0; i < list.customers.length; i++) {
      if (String(list.customers[i].id) === String(id)) return list.customers[i];
    }
    return null;
  }

  async function ensureCategoryFiles_(cat) {
    var data = state.detailCache;
    if (!data || !cat) return [];
    data.loadedCategories = data.loadedCategories || {};
    data.filesByCategory = data.filesByCategory || {};
    if (data.loadedCategories[cat]) return data.filesByCategory[cat] || [];
    var res = await run('api_listCustomerCategoryFiles', data.customer.id, cat);
    var files = (res && res.files) || [];
    data.filesByCategory[cat] = files;
    data.loadedCategories[cat] = true;
    if (!data.folderMeta) data.folderMeta = {};
    if (!data.folderMeta[cat]) data.folderMeta[cat] = { done: false, count: 0 };
    data.folderMeta[cat].count = files.length;
    return files;
  }

  function getGlobalCategories_() {
    return ['保單'];
  }

  function buildDetailFromCache_(id) {
    var lite = findCachedCustomer_(id);
    if (!lite) return null;
    var categories = getGlobalCategories_();
    if (!categories.length) return null;
    var folderMeta = {};
    var src = lite.folderMeta || {};
    categories.forEach(function (cat) {
      folderMeta[cat] = src[cat] ? {
        done: !!src[cat].done,
        count: Number(src[cat].count) || 0
      } : { done: false, count: 0 };
    });
    var filesByCategory = {};
    categories.forEach(function (cat) { filesByCategory[cat] = []; });
    return {
      customer: Object.assign({}, lite, { folderMeta: folderMeta }),
      categories: categories,
      folderMeta: folderMeta,
      filesByCategory: filesByCategory,
      loadedCategories: {},
      filesLoaded: false
    };
  }

  async function renderDetail(content) {
    var tab = state.detailTab || 'folders';
    var seq = ++renderSeq_;
    var t0 = (window.performance && performance.now) ? performance.now() : Date.now();

    // 已有相同客戶的詳情快取 → 完全免打伺服器
    var data = state.detailCache && state.detailCache.customer &&
      String(state.detailCache.customer.id) === String(state.customerId)
      ? state.detailCache : null;
    var cacheSource = data ? 'detailCache'
      : (findCachedCustomer_(state.customerId) ? 'listCache' : 'server');

    if (!data) data = buildDetailFromCache_(state.customerId);
    if (data) state.detailCache = data;
    try {
      console.log('[DriveDocs]', 'renderDetail', {
        id: state.customerId, source: cacheSource,
        hasFolderMeta: !!(data && data.customer && data.customer.folderMeta),
        cats: getGlobalCategories_().length
      });
    } catch (e) { /* ignore */ }

    if (data) {
      if (tab === 'info' || tab === 'notes') {
        try {
          data = await hydrateDetailCustomer_(data, seq);
        } catch (eHydra) { /* 仍顯示列表快取 */ }
        if (seq !== renderSeq_) return;
      }
      paintDetailShell_(content, data.customer, tab, data);
      var body0 = $('detail-body');
      if (!body0) return;
      if (tab === 'folders') renderFolderTab(body0, data);
      else if (tab === 'info') renderInfoTab(body0, data.customer);
      else renderNotesTab(body0, data.customer);
      try {
        var t1 = (window.performance && performance.now) ? performance.now() : Date.now();
        console.log('[DriveDocs]', 'detail painted in', Math.round(t1 - t0), 'ms · via', cacheSource);
      } catch (e2) { /* ignore */ }
      return;
    }

    // fallback：無列表快取（如深連結），才打一次伺服器
    content.innerHTML = '<p class="muted soft-load" style="padding:16px">載入客戶…</p>';
    try {
      data = await run('api_getCustomer', state.customerId, false);
    } catch (err) {
      if (seq !== renderSeq_) return;
      content.innerHTML = '<div class="card__bd"><p>' + escapeHtml(err.message || err) + '</p></div>';
      throw err;
    }
    if (seq !== renderSeq_) return;
    data.loadedCategories = {};
    if (data.customer) data.customer._fullInfo = true;
    state.detailCache = data;
    paintDetailShell_(content, data.customer, tab, data);
    var body = $('detail-body');
    if (!body) return;
    if (tab === 'folders') renderFolderTab(body, data);
    else if (tab === 'info') renderInfoTab(body, data.customer);
    else renderNotesTab(body, data.customer);
  }

  function applyBootPayload_(payload, fromStore) {
    state.app = payload.app || {};
    if (payload.appVersion) state.app.appVersion = payload.appVersion;
    if (payload.activityPathHint) state.app.activityPathHint = payload.activityPathHint;
    if (payload.paths) state.app.paths = payload.paths;
    if (payload.weekLabel) state.weekLabel = payload.weekLabel;
    if (payload.home) rememberHome_(payload.home);
    if (payload.activities) {
      state.cache.activities = payload.activities;
      state.cache.activitiesAt = Date.now();
    }
    if (!fromStore) writeBootStore_(payload);
  }

  async function boot() {
    try {
      try { console.log('[DriveDocs] boot · v260814f'); } catch (e) { /* ignore */ }
      // 立刻顯示介面（不等伺服器）
      state.app = {
        appVersion: '260814f',
        user: { name: '楊以寧', display: '楊以寧', role: '', email: '' },
        settings: { rootFolderName: '千婷-整理客戶資料', categories: DEFAULT_CATEGORIES_CLIENT_.slice() },
        initialized: true
      };
      state.page = 'dashboard';
      paintBootVersion_('載入中');
      showShell_();

      var stored = readBootStore_();
      if (stored && stored.app && stored.home) {
        applyBootPayload_(stored, true);
        bootDriveMeta();
        // 同步立刻畫首頁，零等待
        paintDashboardShell_($('content'), stored.home);
        requestAnimationFrame(function () {
          paintDashboardCustomers_($('content'), stored.home);
          animateBars($('content'));
        });
        refreshBootInBackground_();
        return;
      }

      // 無快取：先畫空骨架，再打 api_boot
      paintDashboardShell_($('content'), {
        dashboard: { totalCustomers: '—', todayOrganized: '—', birthdaysThisWeek: { count: 0, customers: [] } },
        customers: { customers: [], initials: [] }
      });
      await refreshBootInBackground_(true);
    } catch (err) {
      $('boot').classList.remove('hidden');
      $('shell').classList.add('hidden');
      $('boot').innerHTML = '<p class="boot__brand">DriveDocs</p><p class="boot__hint">' + escapeHtml(err.message || err) + '</p>' +
        '<p class="boot__hint">若剛更新檔案，請重新「部署 → 新版本」</p>';
    }
  }

  function refreshBootInBackground_(awaitable) {
    var p = run('api_boot').then(function (fresh) {
      applyBootPayload_(fresh, false);
      bootDriveMeta();
      if (state.page === 'dashboard' || !state.page) {
        state.page = 'dashboard';
        var content = $('content');
        paintDashboardShell_(content, fresh.home || state.cache.home);
        paintDashboardCustomers_(content, fresh.home || state.cache.home);
        animateBars(content);
      }
    }).catch(function (err) {
      return run('api_getAppState').then(function (app) {
        if (!app.initialized) return run('api_initialize');
        return app;
      }).then(function (app) {
        state.app = app;
        return run('api_getHome');
      }).then(function (home) {
        rememberHome_(home);
        bootDriveMeta();
        state.page = 'dashboard';
        var content = $('content');
        paintDashboardShell_(content, home);
        paintDashboardCustomers_(content, home);
        animateBars(content);
      }).catch(function (e2) {
        $('content').innerHTML = '<div class="card card__bd"><p>' + escapeHtml((err && err.message) || e2.message || e2) +
          '</p><p class="upload-hint">請確認已貼上最新 Setting.gs（含 api_boot）並「部署新版本」。版本應顯示 v260814f。</p></div>';
      });
    });
    return p;
  }

  function bindChrome() {
    document.querySelectorAll('.menu__item, .navpill__link, .topbar__nav-desktop [data-page], .bottom-nav [data-page], [data-page]').forEach(function (el) {
      if (el._bound) return;
      el._bound = true;
      if (!el.getAttribute('data-page')) return;
      if (el.closest && el.closest('#content')) return;
      el.addEventListener('click', function () {
        var page = el.getAttribute('data-page');
        setSidebarOpen(false);
        setPage(page);
      });
    });

    function toggleDrawer_() {
      var side = $('sidebar');
      var isOpen = side && !side.hidden && !side.classList.contains('hidden');
      setSidebarOpen(!isOpen);
    }

    var toggle = $('sidebar-toggle');
    if (toggle) toggle.onclick = toggleDrawer_;
    var more = $('btn-more-nav');
    if (more) more.onclick = toggleDrawer_;
    var closeBtn = $('btn-close-drawer');
    if (closeBtn) closeBtn.onclick = function () { setSidebarOpen(false); };
    var backdrop = $('sidebar-backdrop');
    if (backdrop) backdrop.onclick = function () { setSidebarOpen(false); };

    var search = $('global-search');
    if (!search) return;
    var timer;
    search.addEventListener('input', function () {
      clearTimeout(timer);
      var q = search.value.trim();
      timer = setTimeout(function () {
        if (!q) {
          if (state.page === 'search') setPage('dashboard');
          return;
        }
        state.searchQuery = q;
        setPage('search');
      }, 280);
    });
  }
